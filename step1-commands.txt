# Step 1: CD の土台（GCP 側）。上から順に 1行ずつ実行。全て終わったらこのファイルは削除して OK
# 最初にプロジェクト番号を変数に入れる（以降の行が $PN を使う。同じターミナルで続けること）

PN=$(gcloud projects describe tsunagu-works-829174 --format='value(projectNumber)')

# --- A. Cloud Build を有効化し、「ビルドの身分」（compute 既定SA）に必要ロールを付与 ---

gcloud services enable cloudbuild.googleapis.com iamcredentials.googleapis.com

gcloud projects add-iam-policy-binding tsunagu-works-829174 --member="serviceAccount:${PN}-compute@developer.gserviceaccount.com" --role="roles/run.admin"

gcloud projects add-iam-policy-binding tsunagu-works-829174 --member="serviceAccount:${PN}-compute@developer.gserviceaccount.com" --role="roles/artifactregistry.writer"

gcloud projects add-iam-policy-binding tsunagu-works-829174 --member="serviceAccount:${PN}-compute@developer.gserviceaccount.com" --role="roles/logging.logWriter"

gcloud iam service-accounts add-iam-policy-binding ${PN}-compute@developer.gserviceaccount.com --member="serviceAccount:${PN}-compute@developer.gserviceaccount.com" --role="roles/iam.serviceAccountUser"

# --- B. 「GitHub Actions の身分」（github-deployer SA）を作成し、WIF でキーレス連携 ---

gcloud iam service-accounts create github-deployer --display-name="GitHub Actions deployer"

gcloud projects add-iam-policy-binding tsunagu-works-829174 --member="serviceAccount:github-deployer@tsunagu-works-829174.iam.gserviceaccount.com" --role="roles/cloudbuild.builds.editor"

gcloud projects add-iam-policy-binding tsunagu-works-829174 --member="serviceAccount:github-deployer@tsunagu-works-829174.iam.gserviceaccount.com" --role="roles/serviceusage.serviceUsageConsumer"

gcloud projects add-iam-policy-binding tsunagu-works-829174 --member="serviceAccount:github-deployer@tsunagu-works-829174.iam.gserviceaccount.com" --role="roles/viewer"

# ソースアップロード用バケットを先に作り、バケット単位の最小権限を付与（deployer=書き込み・ビルドSA=読み取り）

gcloud storage buckets create gs://tsunagu-works-829174_cloudbuild --location=us-central1

gcloud storage buckets add-iam-policy-binding gs://tsunagu-works-829174_cloudbuild --member="serviceAccount:github-deployer@tsunagu-works-829174.iam.gserviceaccount.com" --role="roles/storage.objectAdmin"

gcloud storage buckets add-iam-policy-binding gs://tsunagu-works-829174_cloudbuild --member="serviceAccount:${PN}-compute@developer.gserviceaccount.com" --role="roles/storage.objectViewer"

# WIF プール + プロバイダ（このリポジトリの GitHub Actions だけを受け入れる条件付き）

gcloud iam workload-identity-pools create github --location=global --display-name="GitHub Actions"

gcloud iam workload-identity-pools providers create-oidc github-provider --location=global --workload-identity-pool=github --display-name="GitHub OIDC" --issuer-uri="https://token.actions.githubusercontent.com" --attribute-mapping="google.subject=assertion.sub,attribute.repository=assertion.repository,attribute.ref=assertion.ref" --attribute-condition="assertion.repository=='masahiro96848/it-matching-service-clean-architecture'"

gcloud iam service-accounts add-iam-policy-binding github-deployer@tsunagu-works-829174.iam.gserviceaccount.com --role="roles/iam.workloadIdentityUser" --member="principalSet://iam.googleapis.com/projects/${PN}/locations/global/workloadIdentityPools/github/attribute.repository/masahiro96848/it-matching-service-clean-architecture"

# --- C. Artifact Registry クリーンアップポリシー（cleanup-policy.json を使用・リポジトリルートで実行） ---

gcloud artifacts repositories set-cleanup-policies tsunagu --location=asia-northeast1 --policy=cleanup-policy.json --no-dry-run

# --- D. 最後に、Step 2 で使う値を表示して Claude に貼り付ける ---

echo "projects/${PN}/locations/global/workloadIdentityPools/github/providers/github-provider"
